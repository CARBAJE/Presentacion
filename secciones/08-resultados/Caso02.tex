\begin{frame}{Resultados — Caso 02: Problema de Sitnikov}
    \textbf{Base del problema:} dos cuerpos masivos en órbita coplanaria y un tercer cuerpo que oscila a lo largo del eje perpendicular (caos determinista clásico).
    \vspace{0.2cm}
    {\small
    \begin{itemize}
        \item Parámetros: $t_{short}=0.5$, $t_{long}=4.0$, $\Delta t=2\times10^{-4}$, integrador=\texttt{IAS15}.
        \item Masas originales (escenario base): $(0.8,\ 1.2,\ 2.5\times10^{-6})$.
        \item Posiciones $\mathbf{r}_0$ (UA): $(-0.5,0,0)$, $(0.5,0,0)$, $(0,0,0.04)$.
        \item Velocidades $\mathbf{v}_0$ (UA/año): $(0,-4.442882938,0)$, $(0,4.442882938,0)$, $(0,0,0.90)$.
    \end{itemize}
    }
\end{frame}
