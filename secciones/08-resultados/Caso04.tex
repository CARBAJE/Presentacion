\subsection{Caso 04: Sistema binario con intruso}
\begin{frame}{Resultados — Caso 04: Sistema binario con intruso}
    \textbf{Descripción:} binaria estelar compacta con un tercer cuerpo (planeta/enano marrón) que ingresa con trayectoria rasante, simulando encuentros en cúmulos abiertos.\\
    \vspace{0.2cm}
    {\small
    \textbf{Configuración del experimento:}
    \begin{itemize}
        \item Horizonte de simulación: $t_{short}=0.6$ años, $t_{long}=3.5$ años.
        \item Integrador: \texttt{IAS15} con paso $\Delta t=2\times10^{-4}$ años.
        \item Masas originales: $(1.25,\ 0.7,\ 0.05)$ M$_\odot$.
        \item Optimización: población 180, 50 épocas, $\omega_{periodicidad}=0.02$.
        \item \textbf{Resultado:} Masas optimizadas $(1.374,\ 0.608,\ 0.263)$ M$_\odot$.
        \item Reducción de caos: $\lambda_{inicial}=-2.81 \rightarrow \lambda_{opt}=-6.01$.
    \end{itemize}
    }
\end{frame}
