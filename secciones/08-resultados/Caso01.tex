\subsection{Caso 01: Problema de Hill}
\begin{frame}{Resultados — Caso 01: Problema de Hill}
    \textbf{Descripción:} simplificación del problema restringido de 3 cuerpos; sistema binario con un cuerpo ligero cercano al radio de Hill.\\
    \vspace{0.2cm}
    {\small
    \textbf{Configuración del experimento:}
    \begin{itemize}
        \item Horizonte de simulación: $t_{short}=0.5$ años, $t_{long}=4.0$ años.
        \item Integrador: \texttt{IAS15} con paso $\Delta t=2\times10^{-4}$ años.
        \item Masas originales: $(1.3,\ 0.8,\ 1.3\times10^{-3})$ M$_\odot$.
        \item Optimización: población 180, 50 épocas, $\omega_{periodicidad}=500.0$.
        \item \textbf{Resultado:} Masas optimizadas $(1.066,\ 0.700,\ 0.0012)$ M$_\odot$.
        \item Reducción de caos: $\lambda_{inicial}=\infty \rightarrow \lambda_{opt}=-3.24$.
    \end{itemize}
    }
\end{frame}
